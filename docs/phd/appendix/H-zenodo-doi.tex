#!/usr/bin/env python3
# verify_doi.py -- checks all 13 Zenodo DOIs resolve
import urllib.request

dois = [
  "10.5281/zenodo.19227877",  "10.5281/zenodo.15193990",
  "10.5281/zenodo.15280831",  "10.5281/zenodo.15377927",
  "10.5281/zenodo.15428791",  "10.5281/zenodo.15491824",
  "10.5281/zenodo.15521093",  "10.5281/zenodo.15553476",
  "10.5281/zenodo.15571829",  "10.5281/zenodo.15589044",
  "10.5281/zenodo.15603271",  "10.5281/zenodo.15618539",
  "10.5281/zenodo.15631084",
]

for doi in dois:
    url = f"https://doi.org/{doi}"
    try:
        req = urllib.request.Request(url)
        urllib.request.urlopen(req, timeout=10)
        print(f"OK  {doi}")
    except Exception as e:
        print(f"ERR {doi}  -- {e}")
\end{verbatim}
