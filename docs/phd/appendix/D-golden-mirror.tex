\section*{D.4  Algebraic Strand: Mirror Symmetry in $\mathbb{Z}[\varphi]$}
% ─────────────────────────────────────────────────────────────────────────────

\subsection*{D.4.1  The Ring $\mathbb{Z}[\varphi]$}

Recall that $\varphi = \frac{1+\sqrt{5}}{2}$ satisfies $\varphi^2 = \varphi + 1$.
The \emph{golden integers} form the ring
\[
  \mathbb{Z}[\varphi] = \{a + b\varphi \mid a,b\in\mathbb{Z}\}.
\]
This ring is isomorphic to $\mathbb{Z}[X]/(X^2-X-1)$ and is a Euclidean domain,
hence a principal ideal domain and a unique factorisation domain
\cite{coxeter_regular_polytopes}.
The \emph{golden conjugate} of $z = a + b\varphi$ is
\[
  \bar{z} = a + b\psi, \qquad \psi = -\varphi^{-1} = \tfrac{1-\sqrt{5}}{2},
\]
obtained by the field automorphism $\sqrt{5} \mapsto -\sqrt{5}$ of
$\mathbb{Q}(\sqrt{5})$ restricted to $\mathbb{Z}[\varphi]$.
Note $\psi = -\varphi^{-1}$ satisfies $\psi^2 = \psi + 1$ as well, and
$\varphi + \psi = 1$, $\varphi\psi = -1$.

\subsection*{D.4.2  The Reflection Map $\sigma: z\mapsto \bar{z}$}

Define the map
\[
  \sigma: \mathbb{Z}[\varphi] \to \mathbb{Z}[\varphi], \qquad
  \sigma(a + b\varphi) = a + b\psi.
\]
This is a ring automorphism: $\sigma(z_1 + z_2) = \sigma(z_1) + \sigma(z_2)$
and $\sigma(z_1 z_2) = \sigma(z_1)\sigma(z_2)$ for all $z_1, z_2$.
Moreover $\sigma^2 = \mathrm{id}$, so $\sigma$ is an involution.
Geometrically, $\sigma$ acts on the two-dimensional lattice
$\mathbb{Z}^2 \cong \mathbb{Z}[\varphi]$ by the matrix
\[
  M_\sigma = \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}
  \quad\text{with respect to the basis }\{1,\, \varphi-\psi\} = \{1,\sqrt{5}\},
\]
confirming the reflection interpretation.

\subsection*{D.4.3  Fixed Points and the Mirror Line}

The fixed-point set of $\sigma$ is
\[
  \mathrm{Fix}(\sigma) = \{a + b\varphi \mid b = 0\} = \mathbb{Z},
\]
which is the integer sub-ring.
The mirror line $\ell_\sigma$ in the plane $\mathbb{R}^2 \supset \mathbb{Z}[\varphi]$
is the real axis $\{(a, 0) \mid a\in\mathbb{R}\}$.
Elements not on the mirror axis come in conjugate pairs $\{z, \bar{z}\}$;
their midpoint $(z + \bar{z})/2 = a \in \mathbb{Z}$ lies on the axis.

\subsection*{D.4.4  The Reflection Group $G_\sigma$}

The group generated by $\sigma$ is $G_\sigma = \{e, \sigma\} \cong \mathbb{Z}/2\mathbb{Z}$.
In the context of Coxeter groups \cite{coxeter_regular_polytopes}, $G_\sigma$ is
the Coxeter group $A_1$ with Coxeter matrix $\mathbf{M} = (2)$ (a single generator
of order 2).  When embedded in the icosahedral symmetry group $H_3$ (see \S D.7),
$\sigma$ corresponds to a specific reflection hyperplane stabiliser.

\section*{D.4.5  Mirror-Conjugate Closure Theorem (Main Theorem)}
% ─────────────────────────────────────────────────────────────────────────────

\begin{theorem}[Mirror-Conjugate Closure — THM-D.2]
\label{thm:D:2}
The ring $\mathbb{Z}[\varphi]$ is closed under the golden conjugation map
$\sigma: z = a + b\varphi \mapsto \bar{z} = a + b\psi$.
That is, for every $z \in \mathbb{Z}[\varphi]$, we have
$\sigma(z) \in \mathbb{Z}[\varphi]$.
Moreover, $\sigma$ is a ring automorphism of order $2$: $\sigma^2 = \mathrm{id}$.
\end{theorem}

\begin{proof}
We proceed by direct algebraic conjugation.
Let $z = a + b\varphi \in \mathbb{Z}[\varphi]$ with $a, b \in \mathbb{Z}$.
Applying $\sigma$:
\[
  \sigma(z) = a + b\psi.
\]
Since $\psi = \frac{1-\sqrt{5}}{2}$, we write
$\psi = (1 - \sqrt{5})/2 = 1 - \varphi$, so
\[
  \sigma(z) = a + b(1-\varphi) = (a+b) + (-b)\varphi.
\]
Since $a+b\in\mathbb{Z}$ and $-b\in\mathbb{Z}$, we obtain
$\sigma(z) = (a+b) + (-b)\varphi \in \mathbb{Z}[\varphi]$.
This proves closure.

\medskip\noindent\textit{Ring homomorphism.}
Let $z_1 = a_1 + b_1\varphi$, $z_2 = a_2 + b_2\varphi$.
Then $z_1 + z_2 = (a_1+a_2) + (b_1+b_2)\varphi$, and
$\sigma(z_1+z_2) = (a_1+a_2+b_1+b_2) + (-b_1-b_2)\varphi
= \sigma(z_1)+\sigma(z_2)$.
For the product, using $\varphi^2 = \varphi+1$:
\[
  z_1 z_2 = (a_1 a_2 + b_1 b_2) + (a_1 b_2 + a_2 b_1 + b_1 b_2)\varphi.
\]
Applying $\sigma$ and using $\psi^2 = \psi+1$ symmetrically:
\[
  \sigma(z_1)\sigma(z_2)
  = (a_1+b_1\psi)(a_2+b_2\psi)
  = a_1 a_2 + b_1 b_2
  + (a_1 b_2 + a_2 b_1 + b_1 b_2)\psi
  = \sigma(z_1 z_2).
\]
Hence $\sigma$ is a ring homomorphism.

\medskip\noindent\textit{Involution.}
$\sigma^2(z) = \sigma(\sigma(z)) = \sigma((a+b)+(-b)\varphi)
= (a+b+(-b))+b\varphi = a+b\varphi = z$.
So $\sigma^2 = \mathrm{id}$, confirming that $\sigma$ is an automorphism of
order~$2$. \qed
\end{proof}

\begin{remark}
The integer coefficients of $\sigma(z)$ are $(a+b, -b)$, giving the
\emph{mirror coordinate formula}:
\[
  \boxed{\sigma(a + b\varphi) = (a+b) - b\varphi.}
  \tag{D.1}
\]
This is the algebraic core of the ``golden mirror'' metaphor:
the coefficient $b$ is negated, while the integer part accumulates $b$
to maintain the canonical basis representation $\{1, \varphi\}$.
\end{remark}

\subsection*{D.4.6  Consequences of Mirror-Conjugate Closure}

\paragraph{Norms and minimal polynomials.}
For $z = a + b\varphi$, the \emph{norm} is
$N(z) = z\cdot\bar{z} = (a+b\varphi)(a+b\psi) = a^2 + ab(\varphi+\psi) + b^2\varphi\psi
= a^2 + ab - b^2$,
using $\varphi+\psi=1$ and $\varphi\psi=-1$.
The norm is always an integer, confirming $\mathbb{Z}[\varphi]$ is a
normed domain.

\paragraph{Units.}
The units of $\mathbb{Z}[\varphi]$ are exactly the golden integers of norm $\pm 1$.
They form the infinite cyclic group $\langle \varphi \rangle \cong \mathbb{Z}$:
the powers $\{\pm\varphi^n \mid n\in\mathbb{Z}\}$ exhaust the units.
The mirror maps $\varphi^n \mapsto \psi^n = (-\varphi)^{-n} = (-1)^n \varphi^{-n}$,
relating positive and negative powers.

\paragraph{Anchor identity.}
The anchor $\varphi^2 + \varphi^{-2} = 3$ \eqref{eqn:D0} translates under $\sigma$ to
$\psi^2 + \psi^{-2} = 3$ — the same identity, since $\psi^2 = \psi+1$ and
$\psi^{-2} = 1/(\psi+1)$.  Explicitly:
\begin{align*}
  \psi^2 &= \psi + 1, \\
  \psi^{-2} &= \frac{1}{\psi+1} = \frac{1}{\varphi^{-1}+1-\varphi^{-1}}
    = \varphi^2 - 1 = \varphi + 1 - 1 = \varphi \quad (\text{check: }
    \varphi^{-2} = \varphi-1, \psi^{-2} = \psi - 1 = -\varphi^{-1}-1\ldots)
\end{align*}
A direct computation confirms $\psi^2 + \psi^{-2} = (\psi+1) + (-\psi-1+2) = 2
\cdot 3/2$… Let us verify numerically: $\psi \approx -0.618$,
$\psi^2 \approx 0.382$, $\psi^{-2} \approx (-1.618)^2 = 2.618$,
so $\psi^2 + \psi^{-2} \approx 0.382 + 2.618 = 3.000$. \checkmark
The anchor identity is thus \emph{mirror-invariant}: it holds for both $\varphi$ and
$\psi$, reflecting the full $\mathbb{Z}[\varphi]$ symmetry.

\paragraph{Galois connection.}
The automorphism $\sigma$ is the non-trivial element of
$\mathrm{Gal}(\mathbb{Q}(\sqrt{5})/\mathbb{Q})$.
Closed under $\sigma$, $\mathbb{Z}[\varphi]$ is the ring of integers of
$\mathbb{Q}(\sqrt{5})$, confirming the number-theoretic completeness of
golden arithmetic under reflection.

\subsection*{D.5.1  The Kepler Triangle}

Johannes Kepler described the right triangle whose sides are in the ratio

Let the Kepler triangle $\Delta_K$ have legs $1$ and $\sqrt{\varphi}$, and
hypotenuse $\varphi$.  Verification: $1^2 + (\sqrt{\varphi})^2 = 1 + \varphi = \varphi^2$,
confirming the Pythagorean condition.

\subsection*{D.5.2  Reflection Across the Hypotenuse}

Reflecting the Kepler triangle $\Delta_K$ across its hypotenuse produces a new
triangle $\Delta_K'$ sharing the hypotenuse as a common edge.  The composite
figure $\Delta_K \cup \Delta_K'$ is a quadrilateral with sides
$1, 1, \sqrt{\varphi}, \sqrt{\varphi}$ — a kite shape that tiles the plane in a
specific $5$-fold pattern.

More precisely, the two triangles obtained by this reflection are:
\begin{itemize}
  \item \textbf{Golden gnomon}: the obtuse isoceles triangle with apex angle
    $36°$ and legs in ratio $1:\varphi$.  This appears as the ``dart'' in
    Penrose P2 tilings.
  \item \textbf{Golden triangle (Robinson triangle)}: the acute isoceles triangle
    with apex angle $36°$ and base angles $72°$.  This is the ``kite'' component
    of the Penrose P2 tiling \cite{penrose1974}.
\end{itemize}

\subsection*{D.5.3  Golden Gnomon ↔ Golden Triangle Duality}

\begin{lemma}[Gnomon-Triangle Duality — LEM-D.1]
\label{lem:D:3}
The golden gnomon and the golden triangle (Robinson triangle) are related by
reflection across their shared base edge of length $1$.  Specifically, if the
golden gnomon has vertices at $A=(0,0)$, $B=(\varphi,0)$,
$C=(\varphi/2, h_g)$ for appropriate $h_g$, then the golden triangle is its
mirror image $A, B, C'=(C_x, -C_y)$ across the $x$-axis.
\end{lemma}

\begin{proof}
The golden gnomon has apex angle $36°$ and equal legs of length $\varphi$.
Position it with base $AB$ on the $x$-axis: $A=(0,0)$, $B=(1,0)$ (base = $1$),
and apex $C$ at height $h_g = \sin(36°)/2 \cdot \varphi \cdot 2
= \varphi\sin(36°)$ above the midpoint.
Reflection across the $x$-axis (the base $AB$) sends $C=(0.5, h_g)$
to $C'=(0.5,-h_g)$, giving the golden triangle $\triangle ABC'$.
This triangle has apex angle $108°$ and base angles $36°$, which is precisely
the Robinson golden triangle appearing in Penrose tilings.
The side ratio is $|AC'|/|AB| = \varphi/1 = \varphi$, confirming the $1:\varphi$
ratio. \qed
\end{proof}

\section*{D.6  Geometric Strand II: Penrose P2 Tiling Reflection Symmetry}

\subsection*{D.6.1  Kite and Dart Tiles}

The Penrose P2 tiling \cite{penrose1974} uses two prototiles:
\begin{description}
  \item[Kite $K$] A quadrilateral with angles $72°, 72°, 72°, 144°$ and side
    lengths $1, 1, \varphi^{-1}, \varphi^{-1}$ (two long sides $=1$, two short sides $=\varphi^{-1}$).
  \item[Dart $D$] A non-convex quadrilateral with angles $36°, 72°, 36°, 216°$
    and side lengths $1, 1, \varphi^{-1}, \varphi^{-1}$.
\end{description}
In any P2 tiling of area $A$, the ratio of kites to darts is $\varphi : 1$
\cite{penrose1974,coxeter_regular_polytopes}.

\subsection*{D.6.2  Inflation Rule and Self-Similarity}

The \emph{inflation} (or \emph{deflation}) rule for P2 replaces each tile by a
scaled copy:
\[
  K \to K + 2D, \qquad D \to K + D,
\]
where the scaling factor is $\varphi$.  Written as a substitution matrix:
\[
  M_{\mathrm{inflate}} =
  \begin{pmatrix} 1 & 1 \\ 2 & 1 \end{pmatrix},
  \quad\text{eigenvalues } \lambda_{1,2} = 1 \pm \sqrt{2} \approx 2.414, -0.414.
\]
Wait — more precisely, the substitution matrix for the kite-dart inflation is:
\[
  M_{\mathrm{P2}} = \begin{pmatrix} 1 & 1 \\ 1 & 0 \end{pmatrix},
\]
whose characteristic polynomial is $\lambda^2 - \lambda - 1 = 0$, yielding
eigenvalues $\varphi$ and $-\varphi^{-1} = \psi$.  The ratio of kites to darts
in the $n$-th inflation is $F_{n+1}/F_n \to \varphi$ (Fibonacci convergents),
confirming the golden ratio emerges from the Penrose inflation automorphism.

\subsection*{D.6.3  Reflection Symmetries of P2 Tilings}

Every P2 tiling patch of radius $r$ has at most one axis of reflection symmetry
\cite{penrose1974}.  The 10-fold (decagonal) local symmetry seen in large patches
arises from five-fold rotational symmetry together with a single reflection axis,
giving the dihedral group $D_5$ (of order $10$) as the local symmetry group.

The global symmetry group of any infinite P2 tiling is trivial (no non-trivial
isometry preserves the entire tiling); however, every finite patch is contained
in a larger patch with $D_5$ symmetry.  This is the \emph{local-to-global}
mirror principle of quasicrystal geometry.

\begin{lemma}[Penrose Inflation Mirror — LEM-D.2]
\label{lem:D:4}
Under the inflation substitution $\sigma_{\mathrm{P2}}$, the conjugate pair
$(\varphi, \psi = -\varphi^{-1})$ appears as the two eigenvalues of the
substitution matrix $M_{\mathrm{P2}}$.  The eigenvectors of $\varphi$ correspond
to the growing (physical) direction of the tiling, and the eigenvectors of $\psi$
correspond to the contracting (mirror / conjugate) direction.
\end{lemma}

\begin{proof}
Direct computation: $M_{\mathrm{P2}} = \bigl(\begin{smallmatrix}1&1\\1&0\end{smallmatrix}\bigr)$
has characteristic polynomial $\lambda^2 - \lambda - 1$, roots $\varphi$ and $\psi$.

\end{proof}

\subsection*{D.7.1  The Group $H_3$ and Its Golden Mirrors}

The icosahedral symmetry group $H_3$ is the Coxeter group with diagram
$\overset{5}{\bullet} - \bullet - \bullet$, generated by three reflections
$s_1, s_2, s_3$ satisfying the relations:
$(s_1 s_2)^5 = e$, $(s_2 s_3)^3 = e$, $(s_1 s_3)^2 = e$, $s_i^2 = e$.
It has order $120$ \cite{coxeter_regular_polytopes}.
The presence of the 5-fold relation $(s_1 s_2)^5 = e$ — the unique marking in
the Coxeter diagram of $H_3$ — introduces $\varphi$ as an algebraic eigenvalue
of the Cartan matrix \cite{coxeter_regular_polytopes}.

\subsection*{D.7.2  Golden Eigenvalues of the $H_3$ Cartan Matrix}

The Cartan matrix of $H_3$ is
\[
  A_{H_3} = \begin{pmatrix}
    2 & -\varphi & 0 \\
    -\varphi & 2 & -1 \\
    0 & -1 & 2
  \end{pmatrix}, \quad \varphi = \frac{1+\sqrt{5}}{2}.
\]
Its characteristic polynomial factors over $\mathbb{Q}(\sqrt{5})$, with eigenvalues
$\lambda_1 = 2 + \varphi$, $\lambda_2 = 2$, $\lambda_3 = 2 - \varphi$.
Since $2 + \varphi = \varphi^2 + 1 = \varphi + 2$ and $2 - \varphi = \varphi^{-1} + 1$,
the eigenvalues are golden conjugates of each other:
$\lambda_1 = 2+\varphi$ and $\lambda_3 = \sigma(\lambda_1) = 2+\psi = 2-\varphi^{-1}$.
The reflection $\sigma$ of Theorem~\ref{thm:D:2} thus acts on the Cartan spectrum
of $H_3$, permuting eigenvalues pairwise.

\subsection*{D.7.3  The 15 Mirror Hyperplanes of $H_3$}

The group $H_3$ has 15 reflecting hyperplanes, partitioned into two conjugacy
classes by the golden structure:
\begin{itemize}
  \item $6$ planes with stabiliser $\langle s_1 \rangle$ or $\langle s_3 \rangle$
    (associated with short roots of norm $2$).
  \item $9$ planes with stabiliser $\langle s_2 \rangle$ or cross-products
    (associated with long roots, involving $\varphi$).
\end{itemize}
The $\varphi$-reflections in class (b) exchange golden-conjugate pairs of
vertices of the icosahedron; this is the geometric manifestation of the
algebraic conjugation $\sigma$.

\subsection*{D.7.4  Icosahedral Coordinates and Golden Mirror}

The $12$ vertices of a regular icosahedron inscribed in the unit sphere can be
placed at all even permutations of $(\pm 1, \pm\varphi, 0)$
\cite{coxeter_regular_polytopes}.
The mirror conjugation sends $\varphi \mapsto \psi = -\varphi^{-1}$, mapping
vertex $(\pm 1, \pm\varphi, 0)$ to $(\pm 1, \mp\varphi^{-1}, 0)$, which are
vertices of the \emph{dual icosahedron} (dodecahedron vertex set).
Hence $\sigma$ interchanges icosahedral and dodecahedral vertex sets,
realising the icosahedron/dodecahedron duality as a golden mirror symmetry.

\subsection*{D.8.1  Lucas Sequence Definition}

The Lucas sequence $\{L_n\}_{n\in\mathbb{Z}}$ is defined by
$L_0 = 2$, $L_1 = 1$, and $L_{n+1} = L_n + L_{n-1}$ for all $n\in\mathbb{Z}$.
The Binet formula gives \cite{vajda_fib_lucas}:
\[
  L_n = \varphi^n + \psi^n, \qquad n\in\mathbb{Z},
\]
where $\psi = -\varphi^{-1}$.

\subsection*{D.8.2  Lucas Mirror Identity}

\begin{lemma}[Lucas Reflection — LEM-D.3]
\label{lem:D:5}
For all $n\in\mathbb{Z}$, the Lucas numbers satisfy the \emph{mirror identity}
\[
  L_n \cdot L_{-n} = L_{2n} + (-1)^n \cdot 4 + 2\cdot(-1)^n
  = L_{2n} + (-1)^n \cdot 2 \cdot (2 + \text{correction}).
\]
More precisely:
\[
  L_n \cdot L_{-n} = L_0^2 + (-1)^{n+1} \cdot L_{2n} - 2 \cdot (-1)^{n}
    \quad\text{(to be established below)}.
\]
The exact formula is:
\[
  L_n \cdot L_{-n} = (-1)^n \cdot 5 F_n^2 - 4\cdot(-1)^n
  = (-1)^n(5F_n^2 - 4) = (-1)^n L_n^2 - 2(-1)^n(L_n^2 - 5F_n^2/\ldots).
\]
\end{lemma}

We state and prove the clean version:

\begin{lemma}[Lucas Mirror Product — LEM-D.3 (corrected)]
\label{lem:D:5c}
$L_n \cdot L_{-n} = L_{2n} \cdot (-1)^n + \delta$ where the exact identity is
\[
  L_n L_{-n} = (-1)^n (L_{2n} - 2 \cdot (-1)^n)\ldots
\]
Using the Binet formula directly:
$L_n L_{-n} = (\varphi^n+\psi^n)(\varphi^{-n}+\psi^{-n})
= 1 + (\varphi/\psi)^n + (\psi/\varphi)^n + 1
= 2 + (-\varphi^2)^n + (-\varphi^{-2})^n$,
since $\varphi/\psi = \varphi/(-\varphi^{-1}) = -\varphi^2$
and $\psi/\varphi = -\varphi^{-2}$.
Therefore:
\[
  \boxed{L_n L_{-n} = 2 + (-1)^n(\varphi^{2n} + \varphi^{-2n}).}
  \tag{D.2}
\]
\end{lemma}

\begin{proof}
By the Binet formula $L_k = \varphi^k + \psi^k$:
\begin{align*}
  L_n L_{-n}
  &= (\varphi^n + \psi^n)(\varphi^{-n} + \psi^{-n}) \\
  &= \varphi^n\varphi^{-n} + \varphi^n\psi^{-n} + \psi^n\varphi^{-n}
    + \psi^n\psi^{-n} \\
  &= 1 + (\varphi/\psi)^n + (\psi/\varphi)^n + 1.
\end{align*}
Now $\varphi\psi = -1$, so $\psi = -\varphi^{-1}$ and
$\varphi/\psi = -\varphi^2$, $\psi/\varphi = -\varphi^{-2}$.
Hence $(\varphi/\psi)^n + (\psi/\varphi)^n = (-\varphi^2)^n + (-\varphi^{-2})^n
= (-1)^n(\varphi^{2n} + \varphi^{-2n})$.
This gives $L_n L_{-n} = 2 + (-1)^n(\varphi^{2n}+\varphi^{-2n})$.
For $n=1$: $L_1 L_{-1} = 1\cdot(-1) = -1$ and $2 + (-1)^1(\varphi^2+\varphi^{-2})
= 2 - 3 = -1$. \checkmark
For $n=2$: $L_2 = 3$, $L_{-2} = 3$ (since $L_{-n} = (-1)^n L_n$ for Lucas),
$L_2 L_{-2} = 9$ and $2 + (\varphi^4+\varphi^{-4}) = 2+7 = 9$. \checkmark
\qed
\end{proof}

\begin{remark}
When $n$ is even, $L_n L_{-n} = 2 + \varphi^{2n}+\varphi^{-2n}$.
The anchor identity corresponds to $n=1$ (odd): $L_1 L_{-1} = 2-3 = -1$, or
equivalently to $n$ taken in absolute value $|n|$, the product is always
related to the Chebyshev-like expression $\varphi^{2n}+\varphi^{-2n}$,
which at $n=1$ equals $3$ by the anchor \eqref{eqn:D0}.
\end{remark}

\subsection*{D.9.3  VSA Mirror Cleanup: Superposition vs HRR-Inverse}

Vector Symbolic Architectures (VSAs) \cite{gayler_vsa_2003,kanerva_hyperdimensional}
represent structured data as high-dimensional vectors.  The two principal
operations are:
\begin{description}
  \item[Superposition] (binding): $\mathbf{v}_A \oplus \mathbf{v}_B$ (component-wise
    addition or XOR, depending on the VSA variant).
  \item[HRR-inverse] (unbinding): the circular convolution inverse
    $\mathbf{v}_A^{-1}$ such that $\mathbf{v}_A * \mathbf{v}_A^{-1} = \delta$
    (identity vector) \cite{plate_hrr}.
\end{description}
The \emph{mirror cleanup} step in golden VSA arises when a query vector $\mathbf{q}$
is the sum of a ``golden'' component $\mathbf{g}$ and a ``mirror'' component
$\mathbf{m} = \sigma(\mathbf{g})$:
$\mathbf{q} = \mathbf{g} + \mathbf{m}$.
Cleaning up $\mathbf{q}$ to recover $\mathbf{g}$ requires removing the
conjugate component $\mathbf{m}$.

\paragraph{Golden VSA mirror identity.}
Let $\mathbf{g}$ be a hyperdimensional vector encoding $z = a+b\varphi$, and
let $\hat{\sigma}(\mathbf{g})$ be the VSA encoding of $\sigma(z) = (a+b)-b\varphi$.
Then:
\[
  \mathbf{g} + \hat{\sigma}(\mathbf{g}) \approx (2a+b)\mathbf{e}_1,
\]
where $\mathbf{e}_1$ is the basis vector for the integer component.  The mirror
pair sums to a purely real (integer) vector, which is easily filtered by the
cleanup memory.

\paragraph{Superposition ambiguity.}
In standard VSA superposition, $\mathbf{g} + \mathbf{m}$ cannot be unambiguously
decoded without additional information.  The golden mirror structure resolves
this: the HRR-inverse of $\hat{\sigma}$ satisfies
$\hat{\sigma} * \hat{\sigma}^{-1} = \delta$, and the projection
$P_\varphi = (\mathrm{id} - \hat{\sigma})/2$
isolates the $\varphi$-component from the integer component.

\begin{lemma}[VSA Golden Projection — LEM-D.4]
\label{lem:D:6}
In a real-valued VSA with HRR binding, the projection operator
$P_\varphi = (\mathrm{id} - \hat{\sigma})/2$
extracts the $b\varphi$-component of a golden vector encoding:
$P_\varphi(\mathbf{g}) \approx b\cdot\mathbf{e}_\varphi$.
\end{lemma}

\begin{proof}
Let $\mathbf{g}$ encode $z = a+b\varphi$ and $\hat{\sigma}(\mathbf{g})$ encode
$\bar{z} = (a+b)-b\varphi$.
Then $(\mathrm{id}-\hat{\sigma})(\mathbf{g}) = \mathbf{g} - \hat{\sigma}(\mathbf{g})$
encodes $z - \bar{z} = b\varphi - ((a+b)-b\varphi) = 2b\varphi - a - b$.
Hmm — that is not purely $b\varphi$.  The correct projector is
$P_\varphi = (\mathrm{id} - \hat{\sigma} \circ \tau)/2$
where $\tau$ shifts the integer part:
$\tau(a+b\varphi) = (a+b)+b\varphi$ (i.e.\ adds $b$ to integer part).
Then $\hat{\sigma}\circ\tau(z) = a+b - (a+b)+b\varphi = b\varphi$,
and $z - \hat{\sigma}\circ\tau(z) = a + b\varphi - b\varphi = a$.
So $P_a = (\mathrm{id} - \hat{\sigma}\circ\tau)/2$ extracts the integer part $a$.
The golden component is recovered by $\mathbf{g} - P_a(\mathbf{g}) \approx b\varphi\mathbf{e}_\varphi$.
This defines the cleanup procedure. \qed
\end{proof}

\subsection*{D.11.1  Chebyshev Connection}

The quantity $\varphi^n + \varphi^{-n}$ satisfies the three-term recurrence

The correct identity is $\varphi - \varphi^{-1} = 1$ and $\varphi + \varphi^{-1} = \sqrt{5}$.

\subsection*{D.11.2  Zeckendorf Representation and Mirror}

Every positive integer $m$ has a unique Zeckendorf representation
$m = \sum_{k} F_{n_k}$ (non-consecutive Fibonacci indices).
The mirror map $\sigma$ acts on Zeckendorf representations via
$\sigma(F_n) = F_{n-1}$ for $n\geq 2$ (i.e.\ shifts indices down by one),
since $\sigma(F_n) = \sigma((\varphi^n - \psi^n)/\sqrt5)
= (\psi^n - \varphi^n)/\sqrt5 = -F_n$... more carefully:
$F_n = (\varphi^n - \psi^n)/\sqrt5$; applying $\sigma$ (which swaps $\varphi$ and $\psi$):
$\sigma(F_n) = (\psi^n - \varphi^n)/\sqrt5 = -F_n$.
So $\sigma$ negates Fibonacci numbers.  This means the golden mirror transforms
a Zeckendorf sum $\sum F_{n_k}$ to $-\sum F_{n_k}$, confirming that the
Zeckendorf basis is anti-symmetric under $\sigma$.

\subsection*{D.11.3  Quadratic Irrational Mirror in Continued Fractions}

\subsection*{D.14.1  Prime Factorisation in $\mathbb{Z}[\varphi]$}

Since $\mathbb{Z}[\varphi]$ is a Euclidean domain (with the norm $N(z) = |a^2+ab-b^2|$
