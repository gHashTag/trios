\section{Thesis: One IP, Three SKUs}
\label{sec:36:thesis}

\paragraph{Claim.}
A single ternary-arithmetic compute IP, parametrized only by tile count
$N \in \{1, 4, 16\}$ and process node, yields three production SKUs
(\textbf{TRI-1 Nano} / \textbf{TRI-1 Mid} / \textbf{TRI-1 Max}) that share
\emph{byte-identical receipt ledger} on a canonical workload. This is the
silicon analogue of Apple's M3 / M3 Pro / M3 Max scaling law, but bound by
formal cross-die determinism (gate TG-TRIAD-X, \S\ref{sec:36:tg-triad-x}).

\paragraph{Falsifier.}
If for any pair $(A,B) \in \{\text{Nano},\text{Mid},\text{Max}\}^2$ the
ledger SHA-256 of the canonical GF16 \texttt{dot4} workload
($(1,2,3,4) \mapsto \mathtt{0x47C0}$) differs by even one bit, the
\enquote{one IP} claim is falsified and Ch.36 is withdrawn.

  \item \textbf{Receipt engine:} BLAKE3-mini RECEIPT signer (L-S13, L-S14)

\section{Same Packet Contract (Rule 8)}
\label{sec:36:packet}

All three SKUs use the same 32-bit packet:
\[
\underbrace{[31{:}28]}_{\text{op}} \;\big|\;
\underbrace{[27{:}26]}_{\text{dst}} \;\big|\;
\underbrace{[25{:}24]}_{\text{src}} \;\big|\;
\underbrace{[23{:}20]}_{\text{lane}} \;\big|\;
\underbrace{[19{:}16]}_{\text{rsvd}} \;\big|\;
\underbrace{[15{:}0]}_{\text{payload}}
\]
No SKU-specific fields. \texttt{TRN\_OP\_RECEIPT} is 32-bit on all dies
(no 64-bit \enquote{long receipt} on Max --- otherwise TG-TRIAD-X breaks).

\subsection{Cross-die gate TG-TRIAD-X}
\label{sec:36:tg-triad-x}

\begin{theorem}[Cross-die ledger determinism, draft]
\label{thm:36:triad-x}
Let $L_{\text{Nano}}, L_{\text{Mid}}, L_{\text{Max}}$ denote the JSONL ledger
produced by running the canonical workload $W^* = ((1,2,3,4) \mapsto \mathtt{0x47C0})$
100 times on each SKU. Then under the shared-IP hypothesis:
\[
\mathrm{SHA256}(L_{\text{Nano}}) =
\mathrm{SHA256}(L_{\text{Mid}}) =
\mathrm{SHA256}(L_{\text{Max}}).
\]
\end{theorem}
\admittedbox{Proof skeleton: induction over packet sequence, using
\texttt{trinity-clara/proofs/igla/tri1\_packet\_det.v} (to be written).
Coq witness must reference the existing receipt determinism lemma
from \texttt{tri1\_receipt\_det.v}.}

\paragraph{Why TG-TRIAD-X matters.}
Without this gate, \enquote{one IP, three SKUs} is a marketing claim. With
it, the entire chapter becomes one falsifiable experiment.

\section{Honest Scope Statement (R5)}
\label{sec:36:scope}

\paragraph{What TRI-1 does on-die.}
MAC array (ternary T-MAC), receipt engine, POST self-test, RING27 scratchpad,
9-instruction ALU.

\paragraph{What TRI-1 does NOT do on-die.}
\begin{itemize}
  \item \textbf{No JEPA-native encoder.} Weights stream from off-chip flash/DRAM.
        Target model: \emph{sub-200M-param} (NOT V-JEPA 2's 1B params on
        die). On-die hosts MAC array only.
  \item \textbf{No on-die radio PHY.} LoRa / Wi-Fi / Ethernet sit on carrier PCB.
  \item \textbf{No on-die TRI settlement.} TRI off-chip (Rule 5 from
        TRI\_NET\_SHUTTLE\_TRIAD).
  \item \textbf{No \texttt{*} operators in synthesizable RTL} (R-SI-1).
\end{itemize}
