% Widen the numberline box for List-of-Theorems entries.
% Default \numberline width is 1.5em which is enough only for one-digit
% chapter.theorem indices; two-digit sub-numbers like 21.20, 30.18 collide

\renewcommand*\l@example[2]{\@dottedtocline{0}{0pt}{4.5em}{#1}{#2}}%

\makeatother
