+/**
+ * Algebraic inverse of `estimateTokensForThreshold`: converts a budget-scale
+ * token limit (raw estimate * safetyMultiplier + fixedOverhead) back to the
+ * raw `estimateTokens()` scale, clamped at zero.
+ *
+ * Budget-scale fields on `ComputedConfig` (`triggerThreshold`,
+ * `maxSummarizationInput`, `minSummarizableTokens`) must never be handed to
+ * raw-scale helpers such as `slidingWindow()` or compared against a raw
+ * `estimateTokens()` result. Every budget -> raw conversion goes through this
+ * one function so the two scales cannot drift apart again.
+ *

+ * zero: nothing fits inside the budget, and the clamp keeps callers from
+ * receiving a negative limit.

+  config: ComputedConfig,

+  )

+/**
+ * Collapse runs of whitespace to single spaces, returning the collapsed
+ * text plus a map from collapsed index -> offset in the input text, so
+ * matches can be reported against real line numbers.

+function collapseWhitespace(text) {
+  let collapsed = ''

+  const n = text.length

+      let j = i
+      while (j < n && /\s/.test(text[j])) j++

+      map.push(i)
+      i = j
+    } else {
+      collapsed += text[i]
+      map.push(i)
+      i++
+    }
+  }
+  return { collapsed, map }

+/** Extract a call's argument text (between the parens) from collapsed text. */
+function callArgumentText(text, openParenIndex) {
+  let depth = 1
+  let i = openParenIndex + 1
+  const start = i

+    const c = text[i]
+    if (c === '(' || c === '[' || c === '{') depth++
+    else if (c === ')' || c === ']' || c === '}') {
+      depth--
+      if (depth === 0) break
+    }
+    i++
+  }
+  return text.slice(start, i)

+/** Split argument text on top-level commas (string-aware). */
+function splitTopLevelArgs(args) {
+  const parts = []

+  let depth = 0
+  let quote = null
+  for (let i = 0; i < args.length; i++) {
+    const c = args[i]
+    if (quote) {
+      cur += c
+      if (c === '\\') {
+        if (i + 1 < args.length) cur += args[i + 1]
+        i++
+      } else if (c === quote) quote = null
+      continue
+    }
+    if (c === "'" || c === '"' || c === '`') {
+      quote = c
+      cur += c
+    } else if (c === '(' || c === '[' || c === '{') {
+      depth++
+      cur += c
+    } else if (c === ')' || c === ']' || c === '}') {
+      depth--
+      cur += c
+    } else if (c === ',' && depth === 0) {
+      parts.push(cur.trim())
+      cur = ''
+    } else {
+      cur += c
+    }
+  }
+  if (cur.trim()) parts.push(cur.trim())
+  return parts
